/**
 * GBCL Auction Day Manager - Google Apps Script Backend
 * Supports Settings Buckets, Category column mapping, Drive Photo Upload & Locks
 */

function doGet(e) {
  try {
    var action = (e && e.parameter && e.parameter.action) ? e.parameter.action : 'getInitialData';
    var callback = (e && e.parameter && e.parameter.callback) ? e.parameter.callback : null;
    
    var payload = {};
    if (e && e.parameter && e.parameter.payload) {
      try {
        payload = JSON.parse(decodeURIComponent(e.parameter.payload));
      } catch (pErr) {
        payload = e.parameter;
      }
    } else if (e && e.parameter) {
      payload = e.parameter;
    }

    var output = handleAction(action, payload);
    var jsonString = JSON.stringify(output);

    if (callback) {
      return ContentService.createTextOutput(callback + '(' + jsonString + ')')
        .setMimeType(ContentService.MimeType.JAVASCRIPT);
    }

    return ContentService.createTextOutput(jsonString)
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    var errObj = { success: false, error: err.toString(), stack: err.stack };
    var callback = (e && e.parameter && e.parameter.callback) ? e.parameter.callback : null;
    if (callback) {
      return ContentService.createTextOutput(callback + '(' + JSON.stringify(errObj) + ')')
        .setMimeType(ContentService.MimeType.JAVASCRIPT);
    }
    return ContentService.createTextOutput(JSON.stringify(errObj))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  try {
    var raw = "";
    if (e.parameter && e.parameter.postData) {
      raw = e.parameter.postData;
    } else if (e.postData && e.postData.contents) {
      raw = e.postData.contents;
    }

    var requestData = JSON.parse(raw);
    var resultData = handleAction(requestData.action, requestData.payload);

    return ContentService.createTextOutput(JSON.stringify(resultData))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function handleAction(action, data) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();

  // 1. GET INITIAL DATA (INCLUDING SETTINGS & BUCKETS)
  if (action === 'getInitialData') {
    const settingsRows = getSheetDataAsObjects(ss.getSheetByName('Settings'));
    let buckets = [
      'Batsman',
      'Middle Order Batsman',
      'Lower Order Batsman',
      'All Rounders',
      'Hard Hitter',
      'Bowlers'
    ];

    // Check Settings sheet for Key = 'Buckets' or 'PlayerBuckets'
    for (let i = 0; i < settingsRows.length; i++) {
      const k = (settingsRows[i].Key || '').toString().trim().toLowerCase();
      if ((k === 'buckets' || k === 'playerbuckets') && settingsRows[i].Value) {
        const parsed = settingsRows[i].Value.toString().split(',').map(s => s.trim()).filter(Boolean);
        if (parsed.length > 0) buckets = parsed;
        break;
      }
    }

    return {
      success: true,
      players: getSheetDataAsObjects(ss.getSheetByName('Players')),
      teams: getSheetDataAsObjects(ss.getSheetByName('Teams')),
      results: getSheetDataAsObjects(ss.getSheetByName('AuctionResults') || ss.getSheetByName('Auction Results')),
      logs: getSheetDataAsObjects(ss.getSheetByName('ActivityLog') || ss.getSheetByName('Activity Log')),
      settings: settingsRows,
      buckets: buckets
    };
  }

  // 2. RECORD AUCTION RESULT
  if (action === 'recordAuctionResult') {
    const lock = LockService.getScriptLock();
    lock.waitLock(15000);
    try {
      const pSheet = ss.getSheetByName('Players');
      const rSheet = ss.getSheetByName('AuctionResults') || ss.getSheetByName('Auction Results');
      const logSheet = ss.getSheetByName('ActivityLog') || ss.getSheetByName('Activity Log');

      const pData = pSheet.getDataRange().getValues();
      const headers = pData[0];
      const pIdIdx = headers.indexOf('PlayerID');
      const statusIdx = headers.indexOf('Status');
      const teamIdx = headers.indexOf('SoldTeamID');
      const priceIdx = headers.indexOf('SoldPrice');

      let rowIndex = -1;
      let prevStatus = 'PENDING';
      let prevTeam = '';
      let prevPrice = 0;

      for (let i = 1; i < pData.length; i++) {
        if (pData[i][pIdIdx].toString() === data.playerID.toString()) {
          rowIndex = i + 1;
          prevStatus = pData[i][statusIdx];
          prevTeam = pData[i][teamIdx];
          prevPrice = pData[i][priceIdx];
          break;
        }
      }

      if (rowIndex === -1) {
        return { success: false, error: 'Player ID not found' };
      }

      pSheet.getRange(rowIndex, statusIdx + 1).setValue(data.status);
      pSheet.getRange(rowIndex, teamIdx + 1).setValue(data.status === 'SOLD' ? data.teamID : '');
      pSheet.getRange(rowIndex, priceIdx + 1).setValue(data.status === 'SOLD' ? Number(data.soldPrice) : 0);

      if (rSheet) {
        rSheet.appendRow([
          'RES-' + new Date().getTime(),
          data.playerID,
          data.playerName,
          data.status,
          data.status === 'SOLD' ? data.teamID : '',
          data.status === 'SOLD' ? data.teamName : '',
          data.status === 'SOLD' ? Number(data.soldPrice) : 0,
          new Date().toISOString()
        ]);
      }

      if (logSheet) {
        logSheet.appendRow([
          'LOG-' + new Date().getTime(),
          new Date().toISOString(),
          'AUCTION_RECORDED',
          `Player ${data.playerID} (${data.playerName}) marked ${data.status} for ${data.soldPrice || 0} to ${data.teamName || 'None'}`,
          JSON.stringify({ prevStatus, prevTeam, prevPrice })
        ]);
      }

      return { success: true, message: 'Result recorded successfully' };
    } finally {
      lock.releaseLock();
    }
  }

  // 3. ADD PLAYER (SAVES BUCKET INTO CATEGORY COLUMN)
  if (action === 'addPlayer') {
    const lock = LockService.getScriptLock();
    lock.waitLock(15000);
    try {
      var pSheet = ss.getSheetByName('Players');
      if (!pSheet) throw new Error("Players sheet not found");

      var finalPhotoURL = '';

      if (data.imageBase64 && typeof data.imageBase64 === 'string' && data.imageBase64.length > 100) {
        try {
          var folderIterator = DriveApp.getFoldersByName("GBCL_Player_Photos");
          var folder = folderIterator.hasNext() 
            ? folderIterator.next() 
            : DriveApp.createFolder("GBCL_Player_Photos");
          
          folder.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

          var base64Content = data.imageBase64;
          var commaIdx = base64Content.indexOf(',');
          if (commaIdx > -1) {
            base64Content = base64Content.substring(commaIdx + 1);
          }

          var decodedBytes = Utilities.base64Decode(base64Content);
          if (decodedBytes && decodedBytes.length > 50) {
            var contentType = data.imageType || 'image/jpeg';
            var safeName = (data.playerID || 'player') + '_' + (data.imageName ? data.imageName.replace(/[^a-zA-Z0-9._-]/g, '') : 'photo.jpg');
            
            var blob = Utilities.newBlob(decodedBytes, contentType, safeName);
            var file = folder.createFile(blob);
            file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

            finalPhotoURL = "https://lh3.googleusercontent.com/d/" + file.getId();
          }
        } catch (driveErr) {
          Logger.log("Drive Image Save Error: " + driveErr.toString());
        }
      }

      // Bucket name is stored directly in the Role column (7th column)
      // Category column (10th column) stores 'Regular', 'Reserve Player', or 'Playing Owner'
      var bucketRole = (data.role || data.bucket || 'Batsman').toString().trim();
      var playerCategory = (data.category || 'Regular').toString().trim();

      pSheet.appendRow([
        data.playerID,
        data.playerName,
        data.mobile,
        data.age || '',
        data.area || '',
        finalPhotoURL,
        bucketRole,
        data.battingStyle || 'Right Hand',
        data.bowlingStyle || 'None',
        playerCategory,
        Number(data.basePrice || 1000),
        'PENDING',
        '',
        0
      ]);

      return { success: true, message: 'Player added successfully', photoURL: finalPhotoURL };
    } finally {
      lock.releaseLock();
    }
  }

  // 4. ADD TEAM
  if (action === 'addTeam') {
    const lock = LockService.getScriptLock();
    lock.waitLock(10000);
    try {
      var tSheet = ss.getSheetByName('Teams');
      if (!tSheet) throw new Error("Teams sheet not found");

      tSheet.appendRow([
        data.teamID,
        data.teamName,
        data.logoURL || '🏏',
        data.ownerName,
        Number(data.totalPurse || 100000)
      ]);

      return { success: true, message: 'Team added successfully' };
    } finally {
      lock.releaseLock();
    }
  }

  return { success: false, error: 'Unknown Action: ' + action };
}

function getSheetDataAsObjects(sheet) {
  if (!sheet) return [];
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return [];
  
  const rows = sheet.getDataRange().getValues();
  const headers = rows[0];
  return rows.slice(1).map(row => {
    let obj = {};
    headers.forEach((h, i) => obj[h] = row[i]);
    return obj;
  });
}
